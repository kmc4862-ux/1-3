RESULTS_DIR = Path("results")

def add_error(step, error_type, message):
    """오류 정보를 errors 리스트에 저장한다."""
    errors.append({
        "step": step,
        "type": error_type,
        "message": message
    })


def print_progress(step, message):
    """진행 상황을 출력한다."""
    print(f"[{step}/3] {message}")


def check_api_keys():
    """필수 API 키가 설정되어 있는지 확인한다."""

    missing_keys = []

    if not OPENAI_API_KEY:
        missing_keys.append("OPENAI_API_KEY")

    if not KAKAO_REST_API_KEY:
        missing_keys.append("KAKAO_REST_API_KEY")

    if missing_keys:
        print("\n[오류] API 키가 설정되지 않았습니다.")
        print("다음 환경변수를 설정해주세요.")

        for key in missing_keys:
            print(f"  - {key}")

        print("\n.env.example 파일을 참고하여 .env 파일을 만들어주세요.")
        return False

    return True

def validate_date(date_string):
    """YYYY-MM-DD 형식의 날짜인지 확인한다."""

    try:
        datetime.strptime(date_string, "%Y-%m-%d")
        return True
    except ValueError:
        return False

def call_openai(prompt, temperature=0.7):
    """
    OpenAI Chat Completions API를 호출한다.
    응답 텍스트를 반환한다.
    """

    headers = {
        "Authorization": f"Bearer {OPENAI_API_KEY}",
        "Content-Type": "application/json"
    }

    data = {
        "model": "gpt-4o-mini",
        "temperature": temperature,
        "messages": [
            {
                "role": "system",
                "content": (
                    "당신은 국내 여행 계획을 도와주는 여행 전문가입니다. "
                    "사용자의 요청에 정확하게 답하고, 사실을 확정할 수 없는 "
                    "행사 정보는 후보 또는 참고 정보라고 명시합니다."
                )
            },
            {
                "role": "user",
                "content": prompt
            }
        ]
    }

    try:
        response = requests.post(
            OPENAI_URL,
            headers=headers,
            json=data,
            timeout=60
        )

        response.raise_for_status()

        result = response.json()

        return result["choices"][0]["message"]["content"]

    except requests.exceptions.Timeout:
        raise RuntimeError("OpenAI API 요청 시간이 초과되었습니다.")

    except requests.exceptions.RequestException as exc:
        raise RuntimeError(
            f"OpenAI API 요청 실패: HTTP {getattr(exc.response, 'status_code', 'N/A')}"
        )

    except (KeyError, IndexError, json.JSONDecodeError) as exc:
        raise RuntimeError(
            f"OpenAI API 응답 파싱 실패: {exc}"
        )

def create_recommendation_prompt(travel_date):
    """여행지 추천용 프롬프트를 생성한다."""

    return f"""
여행 날짜는 {travel_date}입니다.

대한민국 국내 여행지를 하나 추천해주세요.

해당 날짜의 계절적 특징을 고려하여 여행하기 좋은 도시 또는 지역을
하나 선정해주세요.

다음 조건을 반드시 지켜주세요.

1. 반드시 JSON 하나만 출력하세요.
2. Markdown 코드 블록을 사용하지 마세요.
3. JSON 외의 설명을 출력하지 마세요.
4. events는 1~3개의 문자열을 가진 배열이어야 합니다.
5. 행사나 축제의 정확한 개최 여부를 확정할 수 없다면
   "후보", "일정 확인 필요" 등의 표현을 사용하세요.
6. reason은 2~4문장으로 작성하세요.

반드시 아래 형식을 지켜주세요.

{{
  "recommended_city": "도시 또는 지역명",
  "weather": "해당 시기의 일반적인 날씨 요약",
  "events": [
    "행사 또는 축제 후보 1",
    "행사 또는 축제 후보 2"
  ],
  "reason": "추천 근거"
}}
"""


def parse_recommendation(raw_text):
    """
    LLM의 응답을 JSON으로 변환한다.
    """

    try:
        data = json.loads(raw_text)

        required_keys = [
            "recommended_city",
            "weather",
            "events",
            "reason"
        ]

        for key in required_keys:
            if key not in data:
                raise ValueError(f"필수 키가 없습니다: {key}")

        if not isinstance(data["recommended_city"], str):
            raise ValueError("recommended_city는 문자열이어야 합니다.")

        if not isinstance(data["weather"], str):
            raise ValueError("weather는 문자열이어야 합니다.")

        if not isinstance(data["events"], list):
            raise ValueError("events는 배열이어야 합니다.")

        if not isinstance(data["reason"], str):
            raise ValueError("reason은 문자열이어야 합니다.")

        return data

    except (json.JSONDecodeError, ValueError, TypeError) as exc:
        raise ValueError(f"JSON 파싱 실패: {exc}")


def get_recommendation(travel_date):
    """
    여행지 추천을 1회 요청하고,
    JSON 파싱 실패 시 1회 재시도한다.
    """

    prompt = create_recommendation_prompt(travel_date)

    for attempt in range(2):

        try:
            raw_response = call_openai(prompt)
            recommendation = parse_recommendation(raw_response)

            return recommendation

        except Exception as exc:

            if attempt == 0:
                print("  - JSON 파싱 실패. JSON 전용 프롬프트로 재시도합니다.")

                prompt = f"""
여행 날짜는 {travel_date}입니다.

아래 스키마를 반드시 만족하는 JSON만 출력하세요.

{{
  "recommended_city": "string",
  "weather": "string",
  "events": ["string"],
  "reason": "string"
}}

여행하기 좋은 국내 지역 하나를 추천하고,
해당 시기의 일반적인 날씨와 행사/축제 후보를 작성하세요.

Markdown, 설명, 코드 블록 없이 JSON만 출력하세요.
"""

            else:
                raise exc

def search_restaurants(city, limit=5):
    """
    Kakao Local API를 이용하여 해당 지역의 맛집을 검색한다.

    Kakao Local API는 keyword 검색에서 장소명, 주소,
    카테고리, x/y 좌표, place_url 등의 정보를 제공한다.
    """

    headers = {
        "Authorization": f"KakaoAK {KAKAO_REST_API_KEY}"
    }

    params = {
        "query": f"{city} 맛집",
        "size": limit
    }

    try:
        response = requests.get(
            KAKAO_LOCAL_URL,
            headers=headers,
            params=params,
            timeout=30
        )

        if response.status_code in (401, 403):
            add_error(
                "place_search",
                "AUTH_ERROR",
                f"HTTP {response.status_code}"
            )

            print(
                f"  - 오류: 인증 실패({response.status_code}). "
                "Kakao REST API 키 설정을 확인하세요."
            )

            return []

        if response.status_code == 429:
            add_error(
                "place_search",
                "QUOTA_ERROR",
                "HTTP 429 - API quota exceeded"
            )

            print("  - 오류: API 호출 한도를 초과했습니다.")
            return []

        response.raise_for_status()

        data = response.json()

        documents = data.get("documents", [])

        if not documents:
            add_error(
                "place_search",
                "EMPTY_RESULT",
                f"0 results for query={city} 맛집"
            )

            print("  - 검색 결과 0건")
            return []

        restaurants = []

        for item in documents[:limit]:

            restaurant = {
                "name": item.get("place_name", ""),
                "address": (
                    item.get("road_address_name")
                    or item.get("address_name")
                    or ""
                ),
                "category": item.get("category_name", ""),
                "url": item.get("place_url", ""),
                "x": safe_float(item.get("x")),
                "y": safe_float(item.get("y"))
            }

            restaurants.append(restaurant)

        return restaurants

    except requests.exceptions.Timeout:

        add_error(
            "place_search",
            "NETWORK_ERROR",
            "Kakao API request timeout"
        )

        print("  - 오류: Kakao API 요청 시간이 초과되었습니다.")
        return []

    except requests.exceptions.RequestException as exc:

        status_code = getattr(exc.response, "status_code", "N/A")

        add_error(
            "place_search",
            "API_ERROR",
            f"HTTP {status_code}"
        )

        print(f"  - 오류: 장소 검색 API 요청 실패 ({status_code})")
        return []

    except (json.JSONDecodeError, TypeError) as exc:

        add_error(
            "place_search",
            "PARSE_ERROR",
            str(exc)
        )

        print("  - 오류: 장소 검색 결과를 파싱하지 못했습니다.")
        return []


def safe_float(value):
    """좌표 문자열을 숫자로 변환한다."""

    if value is None:
        return None

    try:
        return float(value)
    except (ValueError, TypeError):
        return None

다음 정보를 이용해서 국내 여행 추천 리포트를 Markdown으로 작성해주세요.

여행 날짜:
{travel_date}

1차 여행 추천 JSON:
{recommendation_json}

맛집 검색 결과:

반드시 아래 항목을 포함해주세요.

# {travel_date} 국내 여행 추천 리포트

## 추천 지역
추천 지역명을 작성하세요.

## 행사/축제

|---|---|---|---|

맛집 데이터가 없다면 정확히 다음 문구를 사용하세요.

## 1일 일정 제안

오전 / 오후 / 저녁으로 나누어 현실적인 여행 일정을 제안하세요.

## 오류 요약

오류가 없다면:

- 없음

오류가 있다면 오류 내용을 간단히 정리하세요.

중요:
- 제공된 맛집 데이터에 없는 맛집을 임의로 추가하지 마세요.
- 행사 개최 여부를 확정할 수 없는 경우 확정적으로 표현하지 마세요.
- Markdown 형식으로만 출력하세요.
"""


def generate_final_report(travel_date, recommendation, restaurants):
    """최종 여행 리포트를 생성한다."""

    prompt = create_report_prompt(
        travel_date,
        recommendation,
        restaurants
    )

    try:
        report = call_openai(prompt)

        return report

    except Exception as exc:

        add_error(
            "final_report",
            "LLM_ERROR",
            str(exc)
        )

        return create_fallback_report(
            travel_date,
            recommendation,
            restaurants
        )


def create_fallback_report(travel_date, recommendation, restaurants):
    """최종 LLM 호출 실패 시 사용할 기본 리포트."""

    lines = []

    lines.append(f"# {travel_date} 국내 여행 추천 리포트")
    lines.append("")
    lines.append("## 추천 지역")
    lines.append(recommendation["recommended_city"])
    lines.append("")
    lines.append("## 추천 이유")
    lines.append(recommendation["reason"])
    lines.append("")
    lines.append("## 날씨 요약")
    lines.append(recommendation["weather"])
    lines.append("")
    lines.append("## 행사/축제")

    for event in recommendation["events"]:
        lines.append(f"- {event}")

    lines.append("")
    lines.append("## 맛집 추천")

    if restaurants:
        lines.append("")
        lines.append("| 맛집 | 주소 | 카테고리 | 링크 |")
        lines.append("|---|---|---|---|")

        for restaurant in restaurants:
            lines.append(
                f"| {restaurant['name']} | "
                f"{restaurant['address']} | "
                f"{restaurant['category']} | "
                f"{restaurant['url']} |"
            )

    else:
        lines.append("- 데이터 없음")

    lines.append("")
    lines.append("## 1일 일정 제안")
    lines.append("")
    lines.append("- 오전: 추천 지역의 주요 관광지 방문")
    lines.append("- 오후: 지역 맛집 및 관광지 방문")
    lines.append("- 저녁: 주변 명소 및 야경 감상")
    lines.append("")
    lines.append("## 오류 요약")

    if errors:
        for error in errors:
            lines.append(
                f"- [{error['step']}] "
                f"{error['type']}: {error['message']}"
            )
    else:
        lines.append("- 없음")

    return "\n".join(lines)

def save_results(travel_date, recommendation, restaurants):
    """
    원본 JSON과 Markdown 리포트를 results 폴더에 저장한다.
    """

    RESULTS_DIR.mkdir(exist_ok=True)

    raw_data = {
        "travel_date": travel_date,
        "recommendation": recommendation,
        "restaurants": restaurants,
        "errors": errors
    }

    json_path = RESULTS_DIR / f"{travel_date}_raw.json"
    md_path = RESULTS_DIR / f"{travel_date}_travel_plan.md"

    with open(
        json_path,
        "w",
        encoding="utf-8"
    ) as file:
        json.dump(
            raw_data,
            file,
            ensure_ascii=False,
            indent=2
        )

    return json_path, md_path

def parse_args():
    """CLI 인자를 처리한다."""

    parser = argparse.ArgumentParser(

        required=True,

# ============================================================
# main
# ============================================================

def main():

    args = parse_args()

    if not validate_date(args.date):
        print(
            "\n[오류] 날짜 형식이 올바르지 않습니다."
        )
        print(
            "사용 방법: "
            'python travel_planner.py --date "YYYY-MM-DD"'
        )
        return

    if not check_api_keys():
        return

    travel_date = args.date

    print("\n========================================")
    print("      국내 여행 추천 프로그램")
    print("========================================")
    print(f"여행 날짜: {travel_date}")
    print()

    # --------------------------------------------------------
    # 1단계
    # --------------------------------------------------------

    print_progress(
        1,
        "1차 추천 생성 중(LLM)..."
    )

    try:
        recommendation = get_recommendation(travel_date)

        print(
            f"  - recommended_city: "
            f"{recommendation['recommended_city']}"
        )

    except Exception as exc:

        add_error(
            "recommendation",
            "LLM_ERROR",
            str(exc)
        )

        print(f"  - 오류: {exc}")

        print("\n프로그램을 종료합니다.")
        return

    # --------------------------------------------------------
    # 2단계
    # --------------------------------------------------------

    print_progress(
        2,
        "맛집 검색 중(Kakao Local API)..."
    )

    city = recommendation["recommended_city"]

    restaurants = search_restaurants(city)

    if restaurants:
        print(
            f"  - 맛집 {len(restaurants)}곳 검색 완료"
        )
    else:
        print(
            "  - 맛집 데이터 없음. "
            "다음 단계로 진행합니다."
        )

    # --------------------------------------------------------
    # 3단계
    # --------------------------------------------------------

    print_progress(
        3,
        "최종 여행 리포트 생성 중(LLM)..."
    )

    report = generate_final_report(
        travel_date,
        recommendation,
        restaurants
    )

    # 파일 경로 결정
    json_path, md_path = save_results(
        travel_date,
        recommendation,
        restaurants
    )

    # Markdown 파일 저장
    with open(
        md_path,
        "w",
        encoding="utf-8"
    ) as file:
        file.write(report)

    print("  - 리포트 생성 완료")

    print("\n========================================")
    print("완료!")
    print("========================================")
    print(f"원본 JSON: {json_path}")
    print(f"여행 리포트: {md_path}")

    if errors:
        print(f"\n오류 {len(errors)}건이 기록되었습니다.")
    else:
        print("\n오류 없음")


if __name__ == "__main__":
    main()

국내 여행 추천 프로그램

LLM API와 Kakao Local API를 연동하여 사용자가 입력한 여행 날짜를 기준으로 국내 여행지를 추천하고, 추천 지역의 맛집을 검색한 뒤 최종 여행 리포트를 Markdown 파일로 생성하는 CLI 프로그램입니다.

프로젝트 개요

이 프로젝트에서는 하나의 API만 사용하는 것이 아니라 서로 다른 API에서 얻은 데이터를 다음 단계의 입력으로 전달하는 API 연동 과정을 구현했습니다.

전체 처리 과정은 다음과 같습니다.

여행 날짜 입력
    ↓
OpenAI API
    ↓
여행 지역 / 날씨 / 행사 / 추천 이유 JSON 생성
    ↓
추천 지역 추출
    ↓
Kakao Local API
    ↓
추천 지역 맛집 검색
    ↓
OpenAI API
    ↓
최종 여행 리포트 생성
    ↓
results/ 폴더에 JSON + Markdown 저장

오류 처리

프로그램에서는 외부 API를 호출하는 과정에서 발생할 수 있는 여러 오류를 처리합니다.

API 키 오류

API 키가 설정되어 있지 않으면 프로그램을 실행하지 않고 .env 설정 방법을 안내합니다.

장소 검색 인증 오류

Kakao Local API에서 인증 오류가 발생하면 맛집 데이터를 빈 배열로 처리하고 최종 리포트 생성 단계로 계속 진행합니다.

장소 검색 결과 없음

검색 결과가 0건이어도 프로그램을 중단하지 않습니다.

{
  "step": "place_search",
  "type": "EMPTY_RESULT",
  "message": "0 results for query=..."
}

LLM JSON 파싱 오류

LLM이 JSON 형식이 아닌 응답을 반환하면 JSON만 출력하도록 수정된 프롬프트를 이용하여 1회 재시도합니다.

네트워크 오류

API 요청 중 네트워크 오류나 timeout이 발생하면 오류를 기록하고 필요한 경우 다음 단계로 진행합니다.

REST API 이해

REST API는 HTTP를 통해 서버와 데이터를 주고받는 방식입니다.

이 프로젝트에서는 Kakao Local API에 GET 요청을 보내 장소 데이터를 조회합니다.

GET
https://dapi.kakao.com/v2/local/search/keyword.json


요청에는 검색어와 인증 정보가 포함됩니다.

Kakao Local API는 REST API 키를 다음과 같은 Authorization 헤더에 포함하는 방식으로 인증합니다.

Authorization: KakaoAK REST_API_KEY


응답은 JSON 형식으로 받을 수 있습니다.

GET과 POST

GET은 일반적으로 서버의 데이터를 조회할 때 사용합니다.

예:

GET /v2/local/search/keyword.json


POST는 서버에 새로운 데이터를 전달하거나 작업을 요청할 때 주로 사용합니다.

이 프로젝트에서는 장소 검색에 GET 요청을 사용합니다.

API 연결 과정에서 배운 점

이 프로젝트의 핵심은 API 하나를 호출하는 것이 아니라 API의 결과를 다음 API 또는 처리 단계의 입력으로 사용하는 것입니다.

LLM
↓
structured JSON
↓
recommended_city
↓
Kakao Local API
↓
restaurants
↓
LLM
↓
Markdown report


즉, 서로 다른 데이터 형식을 연결하여 하나의 서비스 흐름을 만드는 과정을 실습했습니다.

API 키 보안 주의사항

API 키는 절대로 다음과 같이 Python 코드에 직접 작성하지 않습니다.

OPENAI_API_KEY = "실제_API_키"


대신 .env 파일을 사용합니다.

OPENAI_API_KEY=실제_API_키
KAKAO_REST_API_KEY=실제_API_키
